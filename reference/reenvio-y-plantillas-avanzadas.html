Es de reenvío solo si T es parámetro de esa misma función plantilla, se deduce en la llamada y no lleva const (también auto&&). No lo son: string&&, const T&&, Pila<T>::push(T&&).

Plantillas listas para copiar

#include <utility>   // std::move, std::forward

// Envoltorio que reenvía un argumento tal como llegó
template <typename T>
void env(T&& x) {
    f(std::forward<T>(x));            // con <T>, una sola vez
}

// Fábrica que reenvía todos (así es make_unique)
template <typename T, typename... Args>
unique_ptr<T> crear(Args&&... args) {
    return unique_ptr<T>(new T(std::forward<Args>(args)...));
}

Concepts (C++20, -std=c++20)

#include <concepts>

template <typename T>
concept Numerico = is_integral_v<T> || is_floating_point_v<T>;

template <typename T>
concept TieneTamano = requires(T c) { c.size(); };   // «esto tiene que compilar»

template <Numerico T>                       T doble(T x)  { return x * 2; }
template <typename T> requires Numerico<T>  T mitad(T x)  { return x / 2; }
auto triple(Numerico auto x)                              { return x * 3; }

template <std::totally_ordered T>           T maximo(T a, T b) { return (a > b) ? a : b; }
