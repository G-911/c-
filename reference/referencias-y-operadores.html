El patrón de la función que modifica

void doblar(int& n) { n = n * 2; }    // n ES la variable de quien llama

int a = 4;
doblar(a);                            // se llama sin nada especial → a vale 8

La misma función con apuntador se ve en la lección 2.2: void doblar(int* n) { *n = *n * 2; } y se llama doblar(&a).

Operadores: plantilla de la clase

class Punto {
private:
    int x, y;
public:
    Punto(int px = 0, int py = 0) : x(px), y(py) {}

    bool operator==(const Punto& otro) const {
        return x == otro.x && y == otro.y;
    }
    bool operator!=(const Punto& otro) const {
        return !(*this == otro);
    }
    Punto operator+(const Punto& otro) const {        // devuelve uno NUEVO
        return Punto(x + otro.x, y + otro.y);
    }
    friend ostream& operator<<(ostream& os, const Punto& p) {
        os << "(" << p.x << ", " << p.y << ")";
        return os;                                     // para encadenar
    }
};

a == b llama a a.operator==(b). cout << a llama a operator<<(cout, a): por eso no es método y va con friend.
