Listas dobles, circulares y copia

Hoja de consulta rápida de las lecciones 3.4 y 3.5.

Lista doble

struct Nodo {
    int dato;
    Nodo* anterior;
    Nodo* siguiente;
    Nodo(int d) : dato(d), anterior(nullptr), siguiente(nullptr) {}
};

class ListaDoble {
private:
    Nodo* cabeza;   // primero
    Nodo* ultimo;   // último
public:
    ListaDoble() : cabeza(nullptr), ultimo(nullptr) {}

    void insertarAlFinal(int valor) {
        Nodo* nuevo = new Nodo(valor);
        if (ultimo == nullptr) { cabeza = nuevo; ultimo = nuevo; return; }
        nuevo->anterior = ultimo;
        ultimo->siguiente = nuevo;
        ultimo = nuevo;
    }

    bool eliminar(int valor) {
        Nodo* actual = cabeza;
        while (actual != nullptr && actual->dato != valor) actual = actual->siguiente;
        if (actual == nullptr) return false;
        if (actual->anterior) actual->anterior->siguiente = actual->siguiente;
        else                  cabeza = actual->siguiente;
        if (actual->siguiente) actual->siguiente->anterior = actual->anterior;
        else                   ultimo = actual->anterior;
        delete actual;
        return true;
    }
    // al revés: for (Nodo* a = ultimo; a != nullptr; a = a->anterior)
};

Regla de oro: si A->siguiente == B, entonces B->anterior == A. Cada flecha que cambias tiene una pareja.

Regla de tres: plantilla para ListaEnlazada

void vaciar() {                                     // lo comparten destructor y operator=
    Nodo* actual = cabeza;
    while (actual != nullptr) {
        Nodo* sig = actual->siguiente;
        delete actual;
        actual = sig;
    }
    cabeza = nullptr;
}

~ListaEnlazada() { vaciar(); }                      // 1. destructor

ListaEnlazada(const ListaEnlazada& otra) : cabeza(nullptr) {   // 2. constructor de copia
    for (Nodo* a = otra.cabeza; a != nullptr; a = a->siguiente)
        insertarAlFinal(a->dato);
}

ListaEnlazada& operator=(const ListaEnlazada& otra) {          // 3. asignación
    if (this == &otra) return *this;                // autoasignación
    vaciar();                                       // fuera lo viejo
    for (Nodo* a = otra.cabeza; a != nullptr; a = a->siguiente)
        insertarAlFinal(a->dato);
    return *this;
}
