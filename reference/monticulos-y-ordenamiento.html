Montículo de máximos en un vector

ReglaQué dice
formaárbol completo: se llena por niveles, de izquierda a derecha
ordencada padre >= sus hijos → el mayor en datos[0]
padre de i(i - 1) / 2
hijos de i2*i + 1 y 2*i + 2 (existen si < datos.size())

«Montículo» (heap, estructura de datos) ≠ «montón» (heap, zona de memoria de new).

void subir(int i) {                     // tras push_back
    while (i > 0 && datos[i] > datos[(i - 1) / 2]) {
        swap(datos[i], datos[(i - 1) / 2]);
        i = (i - 1) / 2;
    }
}

void hundir(int i) {                    // tras poner el último en la raíz
    int n = datos.size();
    while (true) {
        int izq = 2 * i + 1, der = 2 * i + 2, mayor = i;
        if (izq < n && datos[izq] > datos[mayor]) mayor = izq;
        if (der < n && datos[der] > datos[mayor]) mayor = der;
        if (mayor == i) break;
        swap(datos[i], datos[mayor]);
        i = mayor;
    }
}

Los algoritmos en pocas líneas

// INSERCIÓN
for (int i = 1; i < n; i++) {
    int x = v[i], j = i - 1;
    while (j >= 0 && v[j] > x) { v[j + 1] = v[j]; j--; }   // j >= 0 PRIMERO
    v[j + 1] = x;
}

// MERGESORT: [ini, fin)
if (fin - ini <= 1) return;
int medio = ini + (fin - ini) / 2;
mergesort(v, ini, medio); mergesort(v, medio, fin);
mezclar(v, ini, medio, fin);      // vector temp; en empate gana la izquierda

// QUICKSORT (Lomuto): [ini, fin], pivote = v[fin]
int pivote = v[fin], i = ini;
for (int j = ini; j < fin; j++)
    if (v[j] <= pivote) swap(v[i++], v[j]);
swap(v[i], v[fin]);               // i = posición final del pivote
quicksort(v, ini, i - 1); quicksort(v, i + 1, fin);
