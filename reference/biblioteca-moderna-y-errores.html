variant y visit: una cosa u otra

using Figura = variant<Circulo, Rectangulo>;
struct CalculaArea {                                  // un operator() por tipo
    double operator()(const Circulo& c) const    { return 3.14159 * c.radio * c.radio; }
    double operator()(const Rectangulo& r) const { return r.ancho * r.alto; }
};
Figura f = Rectangulo{2, 3};
visit(CalculaArea{}, f);          // 6; si falta un operator(), no compila
f.index();                        // qué tipo guarda (desde 0)
holds_alternative<Circulo>(f);    // ¿guarda un Circulo?
get<Rectangulo>(f);               // el valor; tipo equivocado lanza bad_variant_access
get_if<Rectangulo>(&f);           // apuntador, o nullptr si no es ese tipo

Ninguna vista es dueña. Si el texto o el vector muere (o el vector crece y muda sus datos), la vista cuelga. Úsalas como parámetros, no para guardar.

Copy-and-swap

void intercambiar(Pila& otra) noexcept {
    swap(tope, otra.tope);
    swap(n, otra.n);
}
Pila& operator=(Pila otra) noexcept {   // por VALOR: ya copiado (o movido)
    intercambiar(otra);
    return *this;
}                                       // otra libera lo viejo al morir

Garantía fuerte, a = a sin if, y hace de asignación por copia y por movimiento. No lo combines con operator=(Pila&&): ambiguous overload.

expected: valor o error con motivo

expected<int, string> dividir(int a, int b) {
    if (b == 0) return unexpected("división entre cero");
    return a / b;
}
auto r = dividir(1, 0);
if (r) cout << *r;               // el valor
else   cout << r.error();        // el motivo
r.value_or(-1);                  // valor o -1
r.value();                       // con error: lanza bad_expected_access
if (!r) return unexpected(r.error());   // pasar el error hacia arriba
