Esqueleto de un programa

#include <iostream>
#include <string>
using namespace std;        // cómodo en ejercicios; en proyectos grandes se escribe std::

int main() {
    string nombre;
    cin >> nombre;                           // lee hasta el primer espacio
    cout << "Hola, " << nombre << "\n";
    return 0;
}

Condicionales

if (nota < 0 || nota > 20) {        // primero los inválidos
    cout << "invalida";
} else if (nota >= 10) {            // gana el PRIMERO que sea true
    cout << "aprobado";
} else {
    cout << "reprobado";
}

switch (opcion) {                   // solo int o char
    case 1:  cout << "Agregar";  break;
    case 2:  cout << "Eliminar"; break;
    default: cout << "Inválida"; break;
}

int mayor = (a > b) ? a : b;       // ternario: elegir un valor
