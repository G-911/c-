Deshacer y rehacer con dos pilas

void escribir(const string& palabra) {
    pilaDeshacer.push(texto);                        // guarda cómo estaba
    texto = texto.empty() ? palabra : texto + " " + palabra;
    while (!pilaRehacer.empty()) pilaRehacer.pop();  // lo deshecho ya no vale
}
void deshacer() {
    if (pilaDeshacer.empty()) return;
    pilaRehacer.push(texto);
    texto = pilaDeshacer.top();
    pilaDeshacer.pop();
}
// rehacer(): igual que deshacer() con las dos pilas intercambiadas
