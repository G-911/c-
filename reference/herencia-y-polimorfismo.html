Polimorfismo: sintaxis

EscribesSignifica
Base* p = new Hija();un apuntador a la base puede apuntar a una hija
virtual void m(); (base)p->m() llama a la versión del objeto real, no a la del apuntador
void m() override; (hija)«reemplazo un virtual»; si el nombre o los parámetros no cuadran, no compila
virtual double area() = 0;virtual puro: la clase es abstracta; no hay objetos de ella, solo apuntadores
virtual ~Base() {}delete p con Base* p ejecuta también el destructor de la hija

Plantilla completa

class Figura {                              // base abstracta
protected:
    string nombre;                          // las hijas lo ven
public:
    Figura(string n) : nombre(n) {}
    virtual ~Figura() {}                    // SIEMPRE virtual si hay virtuales
    string getNombre() { return nombre; }
    virtual double area() = 0;              // cada hija pone la suya
};

class Circulo : public Figura {
private:
    double radio;
public:
    Circulo(double r) : Figura("circulo"), radio(r) {}   // base primero
    double area() override { return 3.14159 * radio * radio; }
};

class Rectangulo : public Figura {
private:
    double ancho, alto;
public:
    Rectangulo(double an, double al) : Figura("rectangulo"), ancho(an), alto(al) {}
    double area() override { return ancho * alto; }
};

int main() {
    vector<Figura*> plano = { new Circulo(1), new Rectangulo(3, 4) };
    for (Figura* f : plano) {
        cout << f->getNombre() << ": " << f->area() << "\n";   // polimorfismo
    }
    for (Figura* f : plano) delete f;       // cada new, su delete
    plano.clear();
}
