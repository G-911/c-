De Pila de enteros a Pila<T>

template <typename T>
struct Nodo {
    T dato;
    Nodo* siguiente;
    Nodo(const T& d) : dato(d), siguiente(nullptr) {}
};

template <typename T>
class Pila {
    Nodo<T>* tope;
public:
    Pila() : tope(nullptr) {}
    ~Pila();                       // recorre y borra, igual que la 3.6
    void push(const T& valor);    // int → const T&
    void pop();
    T top() const;                 // int → T
    bool vacia() const;
    int tamano() const;
};

Todo el código de una plantilla va en el .h. Si lo separas en .cpp, el enlazador da undefined reference («referencia … sin definir»).

LlamadaCabeceraDevuelve / hace
