Código completo (la versión del curso)

size_t hashCadena(const string& clave) {
    size_t h = 0;                                   // size_t: sin signo, nunca negativo
    for (char c : clave) h = h * 31 + (unsigned char)c;
    return h;
}

class TablaHash {
private:
    vector<list<pair<string, int>>> cubetas;
    size_t cantidad;
    size_t indiceDe(const string& clave) const { return hashCadena(clave) % cubetas.size(); }
public:
    TablaHash(size_t m = 8) : cubetas(m == 0 ? 1 : m), cantidad(0) {}

    double factorDeCarga() const { return (double)cantidad / cubetas.size(); }

    void insertar(const string& clave, int valor) {
        auto& cubeta = cubetas[indiceDe(clave)];
        for (auto& par : cubeta)
            if (par.first == clave) { par.second = valor; return; }   // actualizar
        cubeta.push_back({clave, valor});
        cantidad++;
        if (factorDeCarga() > 1.0) rehash(cubetas.size() * 2);
    }

    bool buscar(const string& clave, int& valor) const {
        for (const auto& par : cubetas[indiceDe(clave)])
            if (par.first == clave) { valor = par.second; return true; }
        return false;
    }

    bool eliminar(const string& clave) {
        auto& cubeta = cubetas[indiceDe(clave)];
        for (auto it = cubeta.begin(); it != cubeta.end(); ++it)
            if (it->first == clave) { cubeta.erase(it); cantidad--; return true; }
        return false;
    }

    void rehash(size_t nuevas) {
        vector<list<pair<string, int>>> viejas(nuevas == 0 ? 1 : nuevas);
        viejas.swap(cubetas);
        for (auto& cubeta : viejas)
            for (auto& par : cubeta)
                cubetas[indiceDe(par.first)].push_back(par);   // índice con el m NUEVO
    }
};

Un rehash invalida los iteradores, pero no las referencias ni los apuntadores a los elementos. El orden de recorrido no está garantizado. Fuente: cppreference — std::unordered_map.

Errores típicos

  • Hash en int: se desborda y sale negativo → índice negativo. Usa size_t.
  • Rehash que copia cubetas[i] a nuevas[i]: las claves quedan donde buscar no mira.
  • Olvidar la división real: cantidad / cubetas.size() sin (double) da 0 o 1.
  • Contar con el orden de recorrido de un unordered_map.
  • Usar m[clave] solo para preguntar si existe: la crea. Usa find o count.
