Lección 5.1 · Árboles binarios

El nodo

struct NodoArbol {
    int dato;
    NodoArbol* izq;
    NodoArbol* der;
    NodoArbol(int d) : dato(d), izq(nullptr), der(nullptr) {}

Lección 5.2 · Árbol binario de búsqueda

Regla: en cada nodo, todo su subárbol izquierdo es menor y todo su subárbol derecho es mayor. Consecuencia: el inorden sale ordenado.

buscar e insertar

bool buscar(int valor) const {
    NodoArbol* actual = raiz;
    while (actual != nullptr) {
        if (valor == actual->dato) return true;
        actual = (valor < actual->dato) ? actual->izq : actual->der;
    }
    return false;
}

NodoArbol* insertarEn(NodoArbol* n, int valor) {         // devuelve la raíz del subárbol
    if (n == nullptr) return new NodoArbol(valor);
    if (valor < n->dato) n->izq = insertarEn(n->izq, valor);
    else                 n->der = insertarEn(n->der, valor);
    return n;
}
// insertar(v):  if (buscar(v)) return false;  raiz = insertarEn(raiz, v);  return true;

mínimo y máximo

if (raiz == nullptr) throw out_of_range("árbol vacío");
NodoArbol* actual = raiz;
while (actual->izq != nullptr) actual = actual->izq;     // máximo: con der
return actual->dato;

eliminar: los tres casos

El nodo es…Qué se hace
hojadelete; el padre apunta a nullptr
con un hijoel hijo sube a su sitio; delete del nodo
con dos hijoscopiar el dato del sucesor inorden (mínimo del subárbol derecho) y eliminar ese sucesor del subárbol derecho
NodoArbol* eliminarEn(NodoArbol* n, int valor) {
    if (n == nullptr) return nullptr;
    if (valor < n->dato) { n->izq = eliminarEn(n->izq, valor); return n; }
    if (valor > n->dato) { n->der = eliminarEn(n->der, valor); return n; }
    if (n->izq == nullptr) { NodoArbol* h = n->der; delete n; return h; }   // hoja o solo der
    if (n->der == nullptr) { NodoArbol* h = n->izq; delete n; return h; }   // solo izq
    NodoArbol* suc = n->der;                                                 // dos hijos
    while (suc->izq != nullptr) suc = suc->izq;
    n->dato = suc->dato;
    n->der = eliminarEn(n->der, suc->dato);
    return n;
}

Lección 5.3 · Árboles AVL

Regla: en cada nodo, factor = altura(izq) − altura(der) vale −1, 0 o +1. Altura ≤ ~1,44·log₂(n): todo es O(log n) siempre.

Rotación a la derecha (caso izquierda-izquierda)

NodoAVL* rotarDerecha(NodoAVL* y) {
    NodoAVL* x = y->izq;
    NodoAVL* t2 = x->der;
    x->der = y;
    y->izq = t2;
    actualizarAltura(y);       // primero el que bajó
    actualizarAltura(x);
    return x;
}

Rotación a la izquierda (caso derecha-derecha)

NodoAVL* rotarIzquierda(NodoAVL* x) {
    NodoAVL* y = x->der;
    NodoAVL* t2 = y->izq;
    y->izq = x;
    x->der = t2;
    actualizarAltura(x);
    actualizarAltura(y);
    return y;
}

Rotación doble izquierda-derecha

n->izq = rotarIzquierda(n->izq); y luego return rotarDerecha(n);

Rotación doble derecha-izquierda

n->der = rotarDerecha(n->der); y luego return rotarIzquierda(n);

Los cuatro casos

CasoNodoHijo de ese ladoArreglo
izquierda-izquierda+2+1 o 0rotar a la derecha el nodo
derecha-derecha−2−1 o 0rotar a la izquierda el nodo
izquierda-derecha+2−1izquierda el hijo, derecha el nodo
derecha-izquierda−2+1derecha el hijo, izquierda el nodo

balancear e insertar

NodoAVL* balancear(NodoAVL* n) {
    actualizarAltura(n);
    int fb = factorBalance(n);
    if (fb > 1) {
        if (factorBalance(n->izq) < 0) n->izq = rotarIzquierda(n->izq);
        return rotarDerecha(n);
    }
    if (fb < -1) {
        if (factorBalance(n->der) > 0) n->der = rotarDerecha(n->der);
        return rotarIzquierda(n);
    }
    return n;
}

NodoAVL* insertarEn(NodoAVL* n, int valor) {
    if (n == nullptr) return new NodoAVL(valor);
    if (valor < n->dato) n->izq = insertarEn(n->izq, valor);
    else                 n->der = insertarEn(n->der, valor);
    return balancear(n);                      // la única diferencia con el BST
}

En la biblioteca estándar

std::map y std::set «normalmente se implementan como árboles rojo-negro» (cppreference): otro árbol balanceado con rotaciones. Por eso son O(log n) y se recorren en orden.
