#define ESPERAR_VERDAD(cond) \
    anotar((cond), string(__FILE__) + ":" + to_string(__LINE__) + ": ESPERAR_VERDAD(" #cond ") falló")

#define ESPERAR_IGUAL(a, b) esperarIgual((a), (b), #a, #b, __FILE__, __LINE__)

#define ESPERAR_LANZA(expr, Tipo)                                             \
    do {                                                                      \
        bool lanzoBien = false;                                               \
        try { expr; } catch (const Tipo&) { lanzoBien = true; } catch (...) {} \
        anotar(lanzoBien, "ESPERAR_LANZA(" #expr ") falló");                  \
    } while (0)

Reglas de las macros: cada argumento entre paréntesis, cada argumento usado una sola vez, y el trabajo en una función. #define CUADRADO(x) x * x hace que CUADRADO(2 + 3) valga 11.

Pruebas que sirven: una idea por bloque { Pila p; ... }, casos borde (vacía, uno, primero, último) y comparar con un valor esperado, nunca consigo misma. Rompe la clase a propósito: alguna prueba debe fallar.

Corrutinas: std::generator (C++23)

#include <generator>
std::generator<int> contar(int hasta) {
    for (int i = 1; i <= hasta; i++)
        co_yield i;                 // entrega i y se pausa
}
for (int x : contar(3)) ...         // 1, 2, 3

Compila con g++ -std=c++23 (g++ 14 o más nuevo). Perezoso: el cuerpo solo corre cuando se pide el siguiente valor, así que puede ser infinito si lo cortas con std::views::take(n). Palabras clave: co_yield, co_await, co_return.

Módulos (C++20)

export module saludo;               // saludo.cpp
export int doble(int x) { return 2 * x; }

import saludo;                      // main.cpp
import std;                         // C++23
g++ -std=c++23 -fmodules -fsearch-include-path -c bits/std.cc   # una vez, para import std
g++ -std=c++23 -fmodules -c saludo.cpp                         # primero el módulo
g++ -std=c++23 -fmodules -c main.cpp                           # después quien importa
g++ std.o saludo.o main.o -o prog

Estado (g++ 15.2): funciona con -fmodules; CMake 3.28 o más nuevo. Para tareas, sigue con .h/.cpp.
