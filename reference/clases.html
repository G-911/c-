Plantilla de clase

class Cuenta {
private:                           // datos: protegidos
    double saldo;
    int movimientos;

public:                            // lo que se puede pedir desde fuera
    Cuenta(double inicial) : saldo(inicial), movimientos(0) {}   // nacer bien
    ~Cuenta() {}                   // morir limpio (aquí iría el delete de lo que pidió)

    bool retirar(double c) {
        if (c <= 0 || c > saldo) return false;   // la clase protege su regla
        saldo -= c;
        movimientos++;
        return true;
    }
    double getSaldo() { return saldo; }          // getter: leer sin poder cambiar
};
