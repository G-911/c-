Proyecto de varios archivos

// agenda.h — declaraciones
#ifndef AGENDA_H          // o, en su lugar: #pragma once
#define AGENDA_H
#include <string>
bool guardar(const std::string& ruta);   // sin cuerpo, sin using namespace
#endif

// agenda.cpp — definiciones
#include "agenda.h"
bool guardar(const std::string& ruta) { ... }
g++ -std=c++17 -Wall main.cpp agenda.cpp -o prog     # todos los .cpp, ningún .h
g++ -std=c++17 -Wall -c agenda.cpp                    # solo compilar → agenda.o
g++ main.o agenda.o -o prog                           # solo enlazar

Makefile mínimo (las líneas de comando empiezan con TABULADOR)

CXX = g++
CXXFLAGS = -std=c++17 -Wall

prog: main.o agenda.o
	$(CXX) $(CXXFLAGS) main.o agenda.o -o prog

main.o: main.cpp agenda.h
	$(CXX) $(CXXFLAGS) -c main.cpp

agenda.o: agenda.cpp agenda.h
	$(CXX) $(CXXFLAGS) -c agenda.cpp

clean:
	rm -f *.o prog

Uso: make, make clean.

CMakeLists.txt mínimo

cmake_minimum_required(VERSION 3.16)
project(Agenda CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(prog main.cpp agenda.cpp)

Uso: cmake -S . -B build && cmake --build build && ./build/prog

Las cinco funciones especiales (para Pila con nodos)

~Pila();                                   // 1 destructor
Pila(const Pila& otra);                    // 2 constructor de copia   (copia profunda)
Pila& operator=(const Pila& otra);         // 3 asignación de copia
Pila(Pila&& otra) noexcept;                // 4 constructor de movimiento
Pila& operator=(Pila&& otra) noexcept;     // 5 asignación por movimiento

// patrón del 4: robar y dejar al otro vacío
Pila(Pila&& otra) noexcept : tope(otra.tope), n(otra.n) {
    otra.tope = nullptr;
    otra.n = 0;
}

Regla de cinco: si escribes una, piensa en las cinco (escribir el destructor o la copia apaga el movimiento automático). Regla de cero: si tus datos van en vector/string/unique_ptr, no escribas ninguna.
