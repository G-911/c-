Clase Grafo con BFS, DFS y camino más corto

class Grafo {
private:
    int n;
    bool dirigido;
    vector<vector<int>> ady;

    void dfsDesde(int u, vector<bool>& visitado, vector<int>& orden) const {
        visitado[u] = true;
        orden.push_back(u);
        for (int v : ady[u])
            if (!visitado[v]) dfsDesde(v, visitado, orden);
    }
public:
    Grafo(int vertices, bool esDirigido = false)
        : n(vertices), dirigido(esDirigido), ady(vertices) {}

    void agregarArista(int u, int v) {
        ady[u].push_back(v);
        if (!dirigido) ady[v].push_back(u);
    }

    vector<int> bfs(int origen) const {                 // por capas, con cola
        vector<int> orden;
        vector<bool> visitado(n, false);
        queue<int> cola;
        visitado[origen] = true;                       // marcar AL ENCOLAR
        cola.push(origen);
        while (!cola.empty()) {
            int u = cola.front(); cola.pop();
            orden.push_back(u);
            for (int v : ady[u])
                if (!visitado[v]) { visitado[v] = true; cola.push(v); }
        }
        return orden;
    }

    vector<int> dfs(int origen) const {                 // en profundidad, recursivo
        vector<int> orden;
        vector<bool> visitado(n, false);
        dfsDesde(origen, visitado, orden);
        return orden;
    }

    vector<int> caminoMasCorto(int origen, int destino) const {   // menos aristas
        vector<int> padre(n, -1);
        vector<bool> visitado(n, false);
        queue<int> cola;
        visitado[origen] = true;
        cola.push(origen);
        while (!cola.empty()) {
            int u = cola.front(); cola.pop();
            for (int v : ady[u])
                if (!visitado[v]) { visitado[v] = true; padre[v] = u; cola.push(v); }
        }
        if (!visitado[destino]) return {};
        vector<int> camino;
        for (int x = destino; x != -1; x = padre[x]) camino.push_back(x);
        reverse(camino.begin(), camino.end());
        return camino;
    }
};

Dijkstra (pesos ≥ 0)

// ady[u] = pares (vecino, peso)
vector<int> dijkstra(int origen) const {
    vector<int> dist(n, INT_MAX);
    priority_queue<pair<int,int>, vector<pair<int,int>>, greater<pair<int,int>>> pq;   // de mínimos
    dist[origen] = 0;
    pq.push({0, origen});                          // (distancia, vértice)
    while (!pq.empty()) {
        auto [d, u] = pq.top(); pq.pop();
        if (d > dist[u]) continue;                 // entrada vieja
        for (auto [v, peso] : ady[u])
            if (dist[u] + peso < dist[v]) {        // relajar
                dist[v] = dist[u] + peso;
                pq.push({dist[v], v});
            }
    }
    return dist;                                   // INT_MAX = inalcanzable
}

¿Qué algoritmo?

Quiero…UsoCoste
visitar todo lo alcanzableBFS o DFSO(V + E)
el camino con menos aristasBFS + padre[]O(V + E)
contar islasun DFS por cada vértice sin visitarO(V + E)
distancias con pesos ≥ 0DijkstraO((V + E) log V)
distancias con pesos negativosBellman‑Ford (no visto en el curso)O(V · E)

Errores típicos

  • Sin visitado: en un grafo con ciclos, el recorrido no termina (DFS desborda la pila de llamadas).
  • BFS que marca al sacar en vez de al encolar: vértices repetidos.
  • Usar DFS para el camino más corto: encuentra un camino, no el más corto.
  • priority_queue sin greater: es de máximos y Dijkstra saca el más lejano.
  • Dijkstra con pesos negativos: resultado incorrecto.
  • INT_MAX + peso: desbordamiento. Relaja solo desde vértices alcanzados.

Memoria: lista O(V + E) frente a matriz O(V²); el DFS recursivo usa un marco por vértice del camino actual. Ver el mapa de la memoria.
